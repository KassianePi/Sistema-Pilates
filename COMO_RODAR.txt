─────────────────────────────────────────────────────────────────────────────
 COMANDOS ÚTEIS
─────────────────────────────────────────────────────────────────────────────

VER LOGS:

  # Backend (API)
  $ docker compose logs backend -f

  # Frontend (React)
  $ docker compose logs frontend -f

  # MySQL (Banco)
  $ docker compose logs mysql -f

  # Todos (CTRL+C para parar)
  $ docker compose logs -f


PARAR CONTAINERS:

  # Parar (mantém dados)
  $ docker compose down

  # Parar E remover volumes (LIMPA TUDO - CUIDADO!)
  $ docker compose down -v


REINICIAR:

  # Um serviço específico
  $ docker compose restart backend

  # Todos
  $ docker compose restart


REBUILD:

  # Rebuild de uma imagem (sem cache)
  $ docker compose build --no-cache backend

  # Build + restart
  $ docker compose up -d --build backend


ENTRAR NO CONTAINER:

  # No backend (Node.js shell)
  $ docker compose exec backend sh

  # No MySQL (MySQL CLI)
  $ docker compose exec mysql mysql -u pilates_user -p

  # No frontend (Node.js shell)
  $ docker compose exec frontend sh


VER TAMANHO DAS IMAGENS:

  $ docker images | grep pilates


EXECUTAR COMANDO EM CONTAINER:

  # Exemplo: instalar pacote no backend
  $ docker compose exec backend npm install novo-pacote

─────────────────────────────────────────────────────────────────────────────
 TROUBLESHOOTING
─────────────────────────────────────────────────────────────────────────────

❌ "Docker command not found"
   → Instalar Docker Desktop
   → Reiniciar terminal/PowerShell

❌ "Porta 3000 em uso"
   → Mudar em docker-compose.yml: ports: - "3001:3000"

❌ "MySQL not ready"
   → Aguardar 30s: sleep 30 && docker compose ps
   → Ver logs: docker compose logs mysql

❌ "Frontend em branco"
   → Abrir DevTools (F12 no navegador)
   → Ver console para erros de CORS/Network

❌ "API retorna 500"
   → Ver logs: docker compose logs backend -f
   → Verificar variáveis de ambiente: docker compose config

❌ "Build failed"
   → Limpeza: docker compose down -v
   → Rebuild: docker compose build --no-cache

❌ "Containers não iniciam"
   → Verificar: docker compose logs
   → Checar portas: lsof -i :3000 (Mac/Linux)
   → Checar portas: netstat -ano | findstr :3000 (Windows)
