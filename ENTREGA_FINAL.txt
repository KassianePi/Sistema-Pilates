═════════════════════════════════════════════════════════════════════════════
 ⚙️  COMANDOS ÚTEIS
═════════════════════════════════════════════════════════════════════════════

INICIAR:
  $ docker compose up -d --build

PARAR (mantém dados):
  $ docker compose down

PARAR (remove dados):
  $ docker compose down -v

VER STATUS:
  $ docker compose ps

VER LOGS:
  $ docker compose logs -f backend      # Logs do backend
  $ docker compose logs -f mysql         # Logs do MySQL
  $ docker compose logs -f               # Todos os logs

ENTRAR EM CONTAINER:
  $ docker compose exec backend sh       # Shell do backend
  $ docker compose exec mysql bash       # Shell do MySQL

BANCO DE DADOS:
  $ docker compose exec mysql mysql -u pilates_user -p pilates_db

REBUILD UM SERVIÇO:
  $ docker compose up -d --build backend

LIMPAR TUDO:
  $ docker system prune -a
  $ docker volume prune
